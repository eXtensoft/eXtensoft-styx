    class BusinessCapability {
        +String ID
        +String Name
        +Enum Tier [Strategic|Core|Supporting]
        +Integer MaturityLevel [1..5]
        +List~Policy~ Constraints
        +mapToValueStage()
    }

    class ValueStreamStage {
        +String Name
        +String TriggeringEvent
        +String EntranceCriteria
        +String ExitCriteria
        +List~ValueItem~ DeliveredValue
        +requiresCapability()
    }

    class BusinessObject {
        +String ConceptName
        +String Definition
        +List~Attribute~ SimpleAttributes
        +Enum Sensitivity [Public|Private|Restricted]
        +StateModel Lifecycle
    }

    class OrganizationUnit {
        +String UnitName
        +Enum Type [Internal|Partner|Vendor]
        +List~Role~ ActiveRoles
        +ownsCapability()
    }

    class EventNode {
        +Timestamp OccurredAt
        +String OriginatingNodeID
        +JSON Payload
        +Threshold TargetValue
    }

    %% Relationships (The 10K Connective Tissue)
    ValueStreamStage "1" --> "*" BusinessCapability : "Activated By"
    BusinessCapability "1" --> "*" BusinessObject : "Manipulates"
    OrganizationUnit "1" --> "*" BusinessCapability : "Accountable For"
    EventNode "*" --> "1" ValueStreamStage : "Triggers Stage"
    EventNode "*" --> "1" BusinessObject : "Reflects State Change"
